11. API quick reference

// Setup
AnimX.config({ debug: false, autoInit: true });
AnimX.init();

// Animate
AnimX.animate('.el', 'fade-up', { duration: 600, delay: 100 });
AnimX.animate('.el', { from: { opacity: 0 }, to: { opacity: 1 } });

// Stagger & Timeline
AnimX.stagger('.items', 'zoom-in', { stagger: { each: 50 } });
AnimX.timeline().add('.el1', 'fade-in').add('.el2', 'fade-in', '<');

// Text
AnimX.text('.title', { split: 'chars', animation: 'text-rise' });

// Scroll
AnimX.scroll('.card', { animation: 'fade-up', once: true });

// Interactions & Components
AnimX.ripple('.btn', { color: 'rgba(255,255,255,0.3)' });
AnimX.component('.card', 'card-lift');

// Global Control
AnimX.stop();
AnimX.destroy();
